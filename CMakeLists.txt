# ─── Filter sources by module path ───────────────────────────────────────────
macro(filter_sources OUT_VAR PATH_FRAGMENT)
    set(${OUT_VAR} "")
    foreach(f ${OF_VENDOR_SOURCES})
        if(f MATCHES "${PATH_FRAGMENT}")
            list(APPEND ${OUT_VAR} ${f})
        endif()
    endforeach()
endmacro()

# ─── Per-module OBJECT libraries (each with correct include scope) ────────────
function(add_of_module NAME SRCS)
    if(NOT SRCS)
        return()
    endif()
    add_library(${NAME} OBJECT ${SRCS})
    target_include_directories(${NAME} PRIVATE ${ARGN})
    target_link_libraries(${NAME} PRIVATE MPI::MPI_CXX ZLIB::ZLIB)
endfunction()

    $<TARGET_OBJECTS:of_fv>

# ─── Solver binary ───────────────────────────────────────────────────────────
add_executable(simpleFoamExtracted "${SOLVER_DIR}/simpleFoam.C")
target_include_directories(simpleFoamExtracted PRIVATE
    ${INC_BASE} ${INC_FV} ${INC_MT} ${INC_PP} ${INC_MTM} ${INC_MTI}
    ${INC_SAM} ${INC_FVM} ${INC_FVC}
    "${SOLVER_DIR}"
)
target_link_libraries(simpleFoamExtracted PRIVATE
    # --whole-archive ensures all RunTime registrations (functionEntries, turbulence models, etc.)
    # are pulled in. Problematic files referencing unvendored deps are excluded from SRCS_SAM/FVM above.
    -Wl,--whole-archive
    $<TARGET_FILE:of_os>
    openfoam_vendor
    -Wl,--no-whole-archive
    MPI::MPI_CXX ZLIB::ZLIB dl pthread
)

install(TARGETS simpleFoamExtracted DESTINATION bin)
